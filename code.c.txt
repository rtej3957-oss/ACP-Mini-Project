#include <stdio.h>
#include <math.h>
#include<stdlib.h>

#define ROWS 30
#define COLS 60

char canvas[ROWS][COLS];

/* Function Prototypes */
void initializeCanvas();
void displayCanvas();

void drawRectangle(int x, int y, int width, int height);
void drawLine(int x1, int y1, int x2, int y2);
void drawTriangle(int x, int y, int height);
void drawCircle(int xc, int yc, int r);

void deleteRectangle(int x, int y, int width, int height);

int main()
{
    int choice;

    initializeCanvas();

    while(1)
    {
        printf("\n===== 2D GRAPHICS EDITOR =====\n");
        printf("1. Draw Rectangle\n");
        printf("2. Draw Line\n");
        printf("3. Draw Triangle\n");
        printf("4. Draw Circle\n");
        printf("5. Delete Rectangle\n");
        printf("6. Display Canvas\n");
        printf("7. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
            {
                int x,y,w,h;
                printf("Enter x y width height: ");
                scanf("%d%d%d%d",&x,&y,&w,&h);
                drawRectangle(x,y,w,h);
                break;
            }

            case 2:
            {
                int x1,y1,x2,y2;
                printf("Enter x1 y1 x2 y2: ");
                scanf("%d%d%d%d",&x1,&y1,&x2,&y2);
                drawLine(x1,y1,x2,y2);
                break;
            }

            case 3:
            {
                int x,y,h;
                printf("Enter top vertex (x y) and height: ");
                scanf("%d%d%d",&x,&y,&h);
                drawTriangle(x,y,h);
                break;
            }

            case 4:
            {
                int xc,yc,r;
                printf("Enter center(x y) and radius: ");
                scanf("%d%d%d",&xc,&yc,&r);
                drawCircle(xc,yc,r);
                break;
            }

            case 5:
            {
                int x,y,w,h;
                printf("Enter rectangle position and size: ");
                scanf("%d%d%d%d",&x,&y,&w,&h);
                deleteRectangle(x,y,w,h);
                break;
            }

            case 6:
                displayCanvas();
                break;

            case 7:
                return 0;

            default:
                printf("Invalid Choice!\n");
        }
    }

    return 0;
}

/* Initialize canvas */
void initializeCanvas()
{
    int i,j;

    for(i=0;i<ROWS;i++)
    {
        for(j=0;j<COLS;j++)
        {
            canvas[i][j]='_';
        }
    }
}

/* Display canvas */
void displayCanvas()
{
    int i,j;

    printf("\n");
    for(i=0;i<ROWS;i++)
    {
        for(j=0;j<COLS;j++)
        {
            printf("%c",canvas[i][j]);
        }
        printf("\n");
    }
}

/* Draw Rectangle */
void drawRectangle(int x, int y, int width, int height)
{
    int i,j;

    for(i=y;i<y+height && i<ROWS;i++)
    {
        for(j=x;j<x+width && j<COLS;j++)
        {
            if(i==y || i==y+height-1 ||
               j==x || j==x+width-1)
            {
                canvas[i][j]='*';
            }
        }
    }
}

/* Draw Line using DDA */
void drawLine(int x1, int y1, int x2, int y2)
{
    int dx=x2-x1;
    int dy=y2-y1;

    int steps = abs(dx)>abs(dy)?abs(dx):abs(dy);

    float xinc=(float)dx/steps;
    float yinc=(float)dy/steps;

    float x=x1;
    float y=y1;

    int i;

    for(i=0;i<=steps;i++)
    {
        if((int)y>=0 && (int)y<ROWS &&
           (int)x>=0 && (int)x<COLS)
        {
            canvas[(int)y][(int)x]='*';
        }

        x += xinc;
        y += yinc;
    }
}

/* Draw Triangle */
void drawTriangle(int x, int y, int height)
{
    int i;

    drawLine(x,y,x-height,y+height);
    drawLine(x,y,x+height,y+height);
    drawLine(x-height,y+height,
             x+height,y+height);
}

/* Draw Circle */
void drawCircle(int xc, int yc, int r)
{
    int x,y;

    for(y=0;y<ROWS;y++)
    {
        for(x=0;x<COLS;x++)
        {
            int d=(x-xc)*(x-xc)+(y-yc)*(y-yc);

            if(abs(d-r*r)<=r)
            {
                canvas[y][x]='*';
            }
        }
    }
}

/* Delete Rectangle */
void deleteRectangle(int x, int y, int width, int height)
{
    int i,j;

    for(i=y;i<y+height && i<ROWS;i++)
    {
        for(j=x;j<x+width && j<COLS;j++)
        {
            canvas[i][j]='_';
        }
    }
}
